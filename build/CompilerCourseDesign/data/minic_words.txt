# ===========================================================================
# mini-C 语言的全部单词（正则表达式规格文件）
# ---------------------------------------------------------------------------
# 命名规则（任务书 p.2）：
#   名字_编码      -- 该正则式需要生成 DFA 图；下划线后的数字即该单词的编码
#   名字_编码L     -- 该命名对应多个单词，编码从该数值起连续分配（L = Launch）
#   Comment_编码   -- 注释：参与自动机构造，但词法分析时不输出单词编码
#   Keyword编码L   -- 关键字：不构造自动机（与 ID 正则重复），只登记编码
#   名字=          -- 不带编码者视为宏定义，只在正则式中做文本替换
# 约定：空串用 # 表示；+ * 等运算符号作字面量时须写成 \+ \*
# 本文件用行首 # 作为注释
# 编码表详见 AGENT.md §6.2
# ===========================================================================

# ---- 宏定义：不参与自动机 ----
digit=[0-9]
letter=[a-zA-Z]

# ---- 关键字：mini-C 关键字必须小写，编码从 103 起连续分配 ----
# 对应编码：103 else  104 if     105 int   106 float  107 double
#           108 return 109 void  110 do    111 while  112 for
# 说明：任务书 p.7 的关键字表漏印了 do，而老师随堂发布的单词表含 do，故补入（编码 110）
Keyword103L=else | if | int | float | double | return | void | do | while | for

# ---- 注释：// 行注释，不能超过一行，识别后不输出单词编码 ----
# ~eoln 表示"除行结束符以外的任意字符"，是系统预定义的特殊命名
Comment_102=//(~eoln)*

# ---- 数字：整数与小数（小数单独占编码 205，避免与 % 冲突）----
# 注意：小数点必须写成 `\.`。本系统的正则式里 `.` 是**连接运算符**，
#       裸写 `digit+ . digit+` 会被解析成"两个 digit+ 的连接"（即整数），
#       小数规则等于失效 —— 这是阶段 7 联调 mini-C 时暴露出来的真实缺陷。
Num_100=digit+
Num_205=digit+ \. digit+

# ---- 标识符：可含下划线 ----
ID_101=(_|letter)(_|letter|digit)*

# ---- 专用符号：编码从 200 起连续分配 ----
# 关键规则：带 L 后缀的定义，**第 k 个选择项拿到的编码是「基号 + k」（k 从 0 计）**。
#           因此`Symbol200L` 的第一项就是 200，不是 201；编码完全由"书写顺序"决定，
#           想跳过某个号就必须**另起一个定义**，靠注释是跳不过去的。
# 本文件据此拆成三段，让实际编码与 AGENT.md §6.2 的表逐号对齐：
#   201 +   202 -   203 *   204 /                       （Symbol201L，4 项）
#   205 —— 留给"小数 Num"（Num_205），符号段在此断开
#   206 ^   207 .   208 <   209 <=  210 >   211 >=
#   212 ==  213 !=  214 =   215 ;   216 ,   217 ++  218 --
#   219 (   220 )   221 [   222 ]   223 {   224 }        （Symbol206L，19 项）
#   226 %   227 !                                        （各占一条，避开 205）
# 注意 1：% 与 ! 单独成条放在 226 / 227，因为 205 已被"小数 Num"占用；
#         若把 % 留在 200L 那串里，后面的符号会整体前移一位，
#         第 6 项 `^` 就会落到 205 上，与小数 Num **撞码**（AGENT.md §6.2 注 1、§13.21）
# 注意 2：`//` 不再单独占编码：它恒被 Comment_102 匹配，注释优先，永远不会输出符号编码。
#         因此 225 号在本语言里保留不用。
# 注意 3：小数点写作 `\.`、自增写作 `\+\+`、正闭包用到的 `+` 写作 `\+` ——
#         `.` 是连接运算符、`+` 是正闭包运算符、`*` 是闭包运算符，
#         裸写会被当成运算符导致该选择项解析失败（AGENT.md §13.20）
Symbol201L= \+ | - | \* | /
Symbol206L= ^ | \. | < | <= | > | >= | == | != | = | ; | , | \+\+ | -- | \( | \) | \[ | \] | \{ | \}
Percent_226=%
Not_227=!
