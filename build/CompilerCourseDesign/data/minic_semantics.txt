# mini-C 语法树结构定义（语义动作）
# 每行对应一条产生式，行序与 minic_grammar.txt 中产生式出现顺序一致（行号见行尾注释）。
# 格式与书写要点见 AGENT.md §10；与 TINY 的那份保持同一套判读规则：
#   -1 0        单链产生式：直接把唯一的右部项当成本次结果，不额外套结点
#   L c…        新建"左部符号名"结点，c… 是要挂成子结点的右部项下标
#   R n         复用右部第 n 项已有的结点（终端叶结点/关键字结点）
#   R1 0 2      双目运算：复用中间的运算符结点，把左右操作数挂上去
# 共 65 条，与文法文件逐条一致。

L 0                      # 0  program -> definition-list
L 0 1                    # 1  definition-list -> definition-list definition
-1 0                     # 2  definition-list -> definition
-1 0                     # 3  definition -> variable-definition
-1 0                     # 4  definition -> function-definition
L 0 1                    # 5  variable-definition -> type-indicator ID SEMI
L 0 1 3                  # 6  variable-definition -> type-indicator ID LMM NUM RMM SEMI
R0                       # 7  type-indicator -> int
R0                       # 8  type-indicator -> float
R0                       # 9  type-indicator -> double
R0                       # 10 type-indicator -> void
L 0 1 3 5                # 11 function-definition -> type-indicator ID LLM parameters RLM compound-stmt
-1 0                     # 12 parameters -> parameter-list
R0                       # 13 parameters -> void
L 0 2                    # 14 parameter-list -> parameter-list DOU parameter
-1 0                     # 15 parameter-list -> parameter
L 0 1                    # 16 parameter -> type-indicator ID
L 0 1                    # 17 parameter -> type-indicator ID LMM RMM
L 1 2                    # 18 compound-stmt -> LBM local-definitions statement-list RBM
L 0 1                    # 19 local-definitions -> local-definitions variable-definition
L                        # 20 local-definitions -> #
L 0 1                    # 21 statement-list -> statement-list statement
L                        # 22 statement-list -> #
-1 0                     # 23 statement -> expression-stmt
-1 0                     # 24 statement -> compound-stmt
-1 0                     # 25 statement -> condition-stmt
-1 0                     # 26 statement -> dowhile-stmt
-1 0                     # 27 statement -> return-stmt
L 0                      # 28 expression-stmt -> expression SEMI
L                        # 29 expression-stmt -> SEMI
R0 2 4                   # 30 condition-stmt -> if LLM expression RLM statement
R0 2 4 6                 # 31 condition-stmt -> if LLM expression RLM statement else statement
R0 1 4                   # 32 dowhile-stmt -> do statement while LLM expression RLM SEMI
R0                       # 33 return-stmt -> return SEMI
R0 1                     # 34 return-stmt -> return expression SEMI
R1 0 2                   # 35 expression -> variable ASSIGN expression
-1 0                     # 36 expression -> simple-expression
L 0                      # 37 variable -> ID
L 0 2                    # 38 variable -> ID LMM expression RMM
R1 0 2                   # 39 simple-expression -> additive-expression relop additive-expression
-1 0                     # 40 simple-expression -> additive-expression
R0                       # 41 relop -> LTEQ
R0                       # 42 relop -> LT
R0                       # 43 relop -> RT
R0                       # 44 relop -> RTEQ
R0                       # 45 relop -> EQ
R0                       # 46 relop -> NE
R1 0 2                   # 47 additive-expression -> additive-expression addop term
-1 0                     # 48 additive-expression -> term
R0                       # 49 addop -> PLUS
R0                       # 50 addop -> MINUS
R1 0 2                   # 51 term -> term mulop factor
-1 0                     # 52 term -> factor
R0                       # 53 mulop -> MULTIPLY
R0                       # 54 mulop -> DIVIDE
R0                       # 55 mulop -> MOD
R1                       # 56 factor -> LLM expression RLM
-1 0                     # 57 factor -> variable
-1 0                     # 58 factor -> call
R0                       # 59 factor -> NUM
L 0 2                    # 60 call -> ID LLM arguments RLM
-1 0                     # 61 arguments -> argument-list
L                        # 62 arguments -> #
L 0 2                    # 63 argument-list -> argument-list DOU expression
-1 0                     # 64 argument-list -> expression
